double weight = 2.3;

for (int i = 0; i < 10; i = i + 1)
{
    weight = weight + 2.5;
}

while (i == 3)
{
    weight = weight - 2.5;
}

if (i > 4)
{
    weight = weight * 2.5;
} else if (i < 4) 
{
    weight = weight / 2.5;
} else if (i >= 10)
{
    weight = weight * 2.5;
} else if (i <= 10)
{
    weight = weight / 2.5;
} else if (i != 10)
{
    weight = weight - 2.5;
}
